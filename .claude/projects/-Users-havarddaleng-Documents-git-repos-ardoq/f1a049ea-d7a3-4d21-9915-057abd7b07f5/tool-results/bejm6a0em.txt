    13	# @ontology <IRI>  — declares the ontology IRI

    15	# Naming conventions:
    16	# - Every entity has an implicit label equal to its identifier unless @label is declared
    17	# - Unprefixed names (e.g., Application) are in the default namespace (:Application)

    19	# - Concept/class identifiers conventionally begin with an upper-case letter
    20	# - Role/reference/field identifiers conventionally begin with a lower-case letter
