    25	(defn ->anchor-patterns
    26	  "A pattern matched by every entity kind we search over (components and references both have
    27	   an id), used to positively bind ?entity for operators that otherwise wouldn't bind it
    28	   themselves (OPTIONAL/NOT EXISTS/UNION-only groups) - without one, such a group evaluates as
    29	   a single graph-wide check disconnected from any specific candidate, instead of per-entity,
    30	   when it ends up alone as an OR branch.
    31	   The query optimizer should fold this into whatever pattern already resolves ?entity from the
    32	   surrounding query, so resolving the id here too should be cheap."
    33	  []
    34	  '[[?entity :ardoq-common/id _]])

   138	(defn child-count-pseudofield->patterns
   139	  "Unlike the other *->patterns fns, binding must be a variable, not a literal value - the count
   140	   is computed, not looked up, so it can only be projected into a variable."
   141	  [binding]
   142	  [[:where {:select ['?entity [(s/count '?child) binding]]
   143	            ;; We use ->is-component-patterns instead of ->anchor-patterns because only components can have children
   144	            :where (conj (->is-component-patterns)
   145	                         [:optional ['[?child :ardoq-common/parent ?entity]]])
   146	            :group-by ['?entity]}]])

   153	(defn field->patterns [field binding]
   154	  (cond
   155	    ;; Built-in fields which can be looked up directly
   156	    (= "_id" field) (id-field->patterns binding)
   157	    (= "component-key" field) (component-key-field->patterns binding)
   158	    (= "name" field) (name-field->patterns binding)
   159	    (= "description" field) (description-field->patterns binding)
   160	    (= "displayText" field) (display-text-field->patterns binding)
   161	    (= "created" field) (created-field->patterns binding)
   162	    (= "last-updated" field) (last-updated-field->patterns binding)
   163	
   164	    ;; Built-in fields which require traversing to another entity
   165	    (= "created-by" field) (created-by-field->patterns binding)
   166	    (= "last-modified-by" field) (last-modified-by-field->patterns binding)
   167	    (= "rootWorkspace" field) (root-workspace-field->patterns binding)
   168	    (= "parent" field) (parent-field->patterns binding)
   169	    (= "source" field) (source-pseudofield->patterns binding)
   170	    (= "target" field) (target-pseudofield->patterns binding)
   171	
   172	    ;; Pseudofields which require traversing to another entity
   173	    (= query-dsl/pseudofield-tags field) (tags-pseudofield->patterns binding)
   174	    (= query-dsl/pseudofield-type-name field) (type-name-pseudofield->patterns binding)
   175	    (= query-dsl/pseudofield-source-name field) (source-component-name-pseudofield->patterns binding)
   176	    (= query-dsl/pseudofield-target-name field) (target-component-name-pseudofield->patterns binding)
   177	    (= query-dsl/pseudofield-source-type-name field) (source-component-type-pseudofield->patterns binding)
   178	    (= query-dsl/pseudofield-target-type-name field) (target-component-type-pseudofield->patterns binding)
   179	    (= query-dsl/pseudofield-parent-name field) (parent-name-pseudofield->patterns binding)
   180	    (= query-dsl/pseudofield-root-workspace-name field) (root-workspace-name-pseudofield->patterns binding)
   181	    (= query-dsl/pseudofield-created-by-name field) (created-by-name-pseudofield->patterns binding)
   182	    (= query-dsl/pseudofield-last-modified-by-name field) (last-modified-by-name-pseudofield->patterns binding)
   183	
   184	    ;; Pseudofields which require a subquery
   185	    (= "child-count" field) (child-count-pseudofield->patterns binding)
   186	
   187	    :else
   188	    (custom-field->patterns field binding)))

   246	(def ^:private date-value-pattern
   247	  (re-pattern (format "^(\\d{4}-\\d{2}-\\d{2}|now)((?:%s)*)$" query-dsl/offset-part-pattern)))
   248	
   249	(defn- offset->period [[_ sign amount unit]]
   250	  (let [n (cond-> (parse-long amount) (= "-" sign) -)]
   251	    (case unit
   252	      "h" (t/hours n)
   253	      "d" (t/days n)
   254	      "M" (t/months n)
   255	      "y" (t/years n))))
   256	
   257	(defn- apply-offsets [date-time offsets]
   258	  (->> (re-seq query-dsl/offset-part-pattern offsets)
   259	       (sort-by (fn [[_ sign]] (if (= "+" sign) 0 1))) ; match postgres in applying all + offsets first
   260	       (map offset->period)
   261	       (reduce t/plus date-time)))

   263	(defn- date-value->date-time
   264	  "Convert a date value like '2024-01-01+1d-2h' or 'now+3h' into a clj-time DateTime."
   265	  [now value]
   266	  (when (string? value)
   267	    (let [value (query-dsl/patch-minutes-to-months value)
   268	          now (if (query-dsl/relative-hourly-value? value)
   269	                now
   270	                (t/floor now t/day))
   271	          [_ date offsets] (re-matches date-value-pattern value)]
   272	      (cond
   273	        (= "now" date)
   274	        (apply-offsets now offsets)
   275	
   276	        (string? date)
   277	        (try (apply-offsets (time-fmt/parse (time-fmt/formatters :date) date) offsets)
   278	             (catch Exception _ nil))))))

   280	(defn- date-value->comparand
   281	  "A date value resolved against `now`, paired with the precision postgres compares it at: whole
   282	   days for everything except an hour-based relative value - see
   283	   ardoq.search.query.build/parse-regular-or-relative-date."
   284	  [now value]
   285	  (when-let [date-time (date-value->date-time now value)]
   286	    {:date-time date-time
   287	     :precision (if (query-dsl/relative-hourly-value? value) :instant :whole-day)}))
   288	
   289	(defn- ->date-time-literal [date-time]
   290	  (s/cast :xsd/dateTime (time-fmt/unparse (time-fmt/formatters :date-hour-minute-second) date-time)))
   291	
   292	(defn- day-after [date-time]
   293	  (t/plus date-time (t/days 1)))
   294	
   295	(defn greater-date->patterns [field {:keys [date-time precision] :as comparand}]
   296	  (let [?date (gensym "?date")]
   297	    (conj (field->patterns field ?date)
   298	          (if (= :instant precision)
   299	            [:filter (s/> ?date (->date-time-literal date-time))] ; exclusive of the instant itself
   300	            [:filter (s/>= ?date (->date-time-literal (day-after date-time)))])))) ; inclusive of the day after
   301	
   302	(defn less-date->patterns [field {:keys [date-time] :as comparand}]
   303	  (let [?date (gensym "?date")]
   304	    (conj (field->patterns field ?date)
   305	          [:filter (s/< ?date (->date-time-literal date-time))]))) ; exclusive of the bound, day-start or instant alike
   306	
   307	(defn between-date->patterns
   308	  [field from to]
   309	  (let [?date (gensym "?date")]
   310	    (conj (field->patterns field ?date)
   311	          [:filter (s/and
   312	                    (s/>= ?date (->date-time-literal (:date-time from)))
   313	                    (if (= :instant (:precision to))
   314	                      (s/<= ?date (->date-time-literal (:date-time to)))
   315	                      (s/< ?date (->date-time-literal (day-after (:date-time to))))))])))
   316	
   317	(defn equal-date->patterns
   318	  "`equal` matches the window the value names: the day it falls on, or the hour it falls in for an
   319	   hour-based value."
   320	  [field {:keys [date-time precision]}]
   321	  (let [?date (gensym "?date")
   322	        [lower span] (if (= :instant precision)
   323	                       [(t/floor date-time t/hour) (t/hours 1)]
   324	                       [date-time (t/days 1)])]
   325	    (conj (field->patterns field ?date)
   326	          [:filter (s/and (s/>= ?date (->date-time-literal lower))
   327	                          (s/< ?date (->date-time-literal (t/plus lower span))))])))

   329	(defn date-rule->patterns
   330	  [now {:keys [field operator value] :as advanced-search-rule}]
   331	  (let [[from to :as comparands] (map (partial date-value->comparand now)
   332	                                      (if (= "between" operator)
   333	                                        [(first value) (second value)]
   334	                                        [value]))]
   335	    (cond
   336	      (= "is_empty" operator)
   337	      (is-empty->patterns field)
   338	
   339	      (= "is_not_empty" operator)
   340	      (is-not-empty->patterns field)
   341	
   342	      (some nil? comparands)
   343	      (error "Unsupported date value" advanced-search-rule)
   344	
   345	      (= "equal" operator)
   346	      (equal-date->patterns field from)
   347	
   348	      (= "greater" operator)
   349	      (greater-date->patterns field from)
   350	
   351	      (= "less" operator)
   352	      (less-date->patterns field from)
   353	
   354	      (= "between" operator)
   355	      (between-date->patterns field from to)
   356	
   357	      :else
   358	      (error "Unsupported date operator" advanced-search-rule))))

   424	(defn reference-type-filter-rule->patterns [{:keys [referenceTypeName value] :as advanced-search-rule}]
   425	  (let [reference-operator (query-dsl/clause->reference-operator advanced-search-rule)]
   426	    (cond
   427	      (= "has-reference-to-component" reference-operator)
   428	      (has-reference-with-type-to-component->patterns value referenceTypeName)
   429	
   430	      (= "has-reference-to-component-type" reference-operator)
   431	      (has-reference-with-type-to-component-type->patterns value referenceTypeName)
   432	
   433	      (= "has-reference-from-component" reference-operator)
   434	      (has-reference-with-type-from-component->patterns value referenceTypeName)
   435	
   436	      (= "has-reference-from-component-type" reference-operator)
   437	      (has-reference-with-type-from-component-type->patterns value referenceTypeName)
   438	
   439	      (= "has-no-reference-to-component" reference-operator)
   440	      (negate (has-reference-with-type-to-component->patterns value referenceTypeName))
   441	
   442	      (= "has-no-reference-to-component-type" reference-operator)
   443	      (negate (has-reference-with-type-to-component-type->patterns value referenceTypeName))
   444	
   445	      (= "has-no-reference-from-component" reference-operator)
   446	      (negate (has-reference-with-type-from-component->patterns value referenceTypeName))
   447	
   448	      (= "has-no-reference-from-component-type" reference-operator)

   498	(defn ->decimal-literal
   499	  "graphlake-sync writes all numbers as decimals. Integers are written without fractional parts."
   500	  [value]
   501	  (if (number? value)
   502	    (let [bd (bigdec value)]
   503	      ;; Strip 0 fractional parts, so 1.0 becomes 1, to match graphlake-sync
   504	      (if (zero? (.remainder bd (bigdec 1)))
   505	        (.setScale bd 0)
   506	        bd))
   507	    value))

   511	(defn and->patterns
   512	  [now {:keys [rules] :as advanced-search-and-rule}]
   513	  (reduce (fn [acc rule]
   514	            (let [result-or-error (advanced-search->sparql-patterns now rule)]
   515	              (if (error? result-or-error)
   516	                (reduced result-or-error)
   517	                (into acc result-or-error))))
   518	          []
   519	          rules))

   521	(defn or->patterns
   522	  [now {:keys [rules] :as advanced-search-or-rule}]
   523	  (reduce (fn [[acc] rule]
   524	            (let [result-or-error (advanced-search->sparql-patterns now rule)]
   525	              (if (error? result-or-error)
   526	                (reduced result-or-error)
   527	                [(conj acc result-or-error)])))
   528	          [[:union]]
   529	          rules))

   531	(defn advanced-search->sparql-patterns
   532	  [now {:keys [condition field operator type value referenceTypeName] :as advanced-search-rule}]
   533	  (let [value (if (= "number" type) (->decimal-literal value) value)]
   534	    (cond
   535	      (= "AND" condition)
   536	      (and->patterns now advanced-search-rule)
   537	
   538	      (= "OR" condition)
   539	      (or->patterns now advanced-search-rule)
   540	
   541	      ;; Specialized rules for handling certain field, operator and/or type combinations
   542	
   543	      (= ["type" "equal" "component"] [field operator value])
   544	      (->is-component-patterns)
   545	
   546	      (= ["type" "equal" "reference"] [field operator value])
   547	      (->is-reference-patterns)
   548	
   549	      (= ["type" "equal" "workspace"] [field operator value])
   550	      (->is-workspace-patterns)
   551	
   552	      (= query-dsl/lucidchart-documents-field field)
   553	      (error "Lucidchart document filtering is not supported in GraphLake advanced search"
   554	             advanced-search-rule)
   555	
   556	      (= "date" type)
   557	      (date-rule->patterns now advanced-search-rule)
   558	
   559	      (= ["equal" "child-count"] [operator field])
   560	      (child-count-equal->patterns value)
   561	
   562	      (= ["not_equal" "child-count"] [operator field])
   563	      (child-count-not-equal->patterns value)
   564	
   565	      (and (query-dsl/has-reference--ids field) referenceTypeName)
   566	      (reference-type-filter-rule->patterns advanced-search-rule)
   567	
   568	      (query-dsl/reference-filter-clause? advanced-search-rule)
   569	      (reference-filter-rule->patterns advanced-search-rule)
   570	
   571	      ;; Standard handling of operators
   572	
   573	      (= "equal" operator)
   574	      (equal->patterns field value)
   575	
   576	      (= "greater" operator)
   577	      (greater->patterns field value)
   578	
   579	      (= "less" operator)
   580	      (less->patterns field value)
   581	
   582	      (= "contains_substring" operator)
   583	      (contains-substring->patterns field value)
   584	
   585	      (= "not_equal" operator)
   586	      (not-equal->patterns field value)
   587	
   588	      (= "not_contains_substring" operator)
   589	      (not-contains-substring->patterns field value)
   590	
   591	      (= "is_empty" operator)
   592	      (is-empty->patterns field)
   593	
   594	      (= "is_not_empty" operator)
   595	      (is-not-empty->patterns field)
   596	
   597	      ;; TODO: support all other operators
   598	
   599	      :else
   600	      (error "Unknown operator"
   601	             advanced-search-rule))))

   657	  [now advanced-search-query]
   658	  (let [patterns (advanced-search->sparql-patterns now advanced-search-query)]
   659	    (if (error? patterns)
   660	      patterns
   661	      {:prefixes prefixes
   662	       :select '[?entityId ?field ?value]
   663	       :where (into patterns
   664	                    ['[?entity :ardoq-common/id ?entityId]
   665	                     [:union component-field-patterns custom-field-patterns]])})))
